# streaming-partial Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A runnable demo where a gsx page streams a skeleton layout, flushes it, then fills the panels **out of document order** using processing-instruction markers — proving what markers add over a plain flush.

**Architecture:** One Go module in a new sibling repo, bootstrapped with `gsx init` (Vite plugin, `gsx dev` HMR, `main.go` on `github.com/gsxhq/vite` serving embedded `dist/`) then layered with `gsxui init`/`add`. The page is a single gsx render: shell + three `<?start name=…>skeleton<?end>` regions, a flush node, then a stream node that blocks, emitting a `<template for="…">` patch and flushing as each panel's data arrives. gsxui supplies the components, vendored the real way.

**Tech Stack:** Go 1.26.1, gsx (processing instructions), gsxui (vendored components), `template-for-polyfill` (vendored, 2.5 KB), Tailwind v4 for a one-off committed CSS build.

```go
package main

import (
	"context"
	"errors"
	"io"
	"net/http"

	"github.com/gsxhq/gsx"
)

// Flush streams whatever has been written so far to the client. It writes no
// markup of its own.
//
// http.NewResponseController walks the writer's Unwrap() chain and prefers a
// FlushError() error over the legacy http.Flusher, so a buffering middleware is
// transparent. A writer that cannot flush reports ErrNotSupported, which is a
// best-effort no-op — that is what lets the page render into a bytes.Buffer in
// tests.
func Flush() gsx.Node {
	return gsx.Func(func(ctx context.Context, w io.Writer) error {
		rw, ok := w.(http.ResponseWriter)
		if !ok {
			return nil
		}
		if err := http.NewResponseController(rw).Flush(); err != nil && !errors.Is(err, http.ErrNotSupported) {
			return err
		}
		return nil
	})
}
```

### Task 5: The streaming server

**Files:**
- Create: `streaming-partial/main.go`, `streaming-partial/data.go`

**Interfaces:**
- Consumes: `views.*` (Tasks 3-4), `Flush` (Task 2).
- Produces:

  - `func (s *server) handle(w http.ResponseWriter, r *http.Request)`

- [ ] **Step 1: Extend the scaffolded main.go**

Task 1's `gsx init` already wrote a `main.go` that builds a `*vite.Vite`, embeds

The shape to add:

```go
package main

import (
	"context"
	"io"
	"log"
	"net/http"
	"time"

	"github.com/gsxhq/gsx"
	"github.com/gsxhq/gsx-examples/streaming-partial/views"
)

type server struct{ latency map[string]time.Duration }

func newServer() *server {
	// Deliberately inverted against document order: the FIRST panel is the
	// slowest, so the page visibly fills out of order.
	return &server{latency: map[string]time.Duration{
		"revenue":  2400 * time.Millisecond,
		"orders":   400 * time.Millisecond,
		"products": 1200 * time.Millisecond,
	}}
}

// stream emits one <template for="…"> per panel, in completion order, flushing
// after each so the browser applies it immediately. It blocks until every panel
// has been sent, which is what keeps the response open.
func (s *server) stream(ctx context.Context) gsx.Node {
	return gsx.Func(func(ctx context.Context, w io.Writer) error {
		results := make(chan views.Panel, len(views.PanelNames()))
		for _, name := range views.PanelNames() {
			go func(name string) {
				select {
				case <-time.After(s.latency[name]):
					results <- panelData(name)
				case <-ctx.Done():
				}
			}(name)
		}
		for range views.PanelNames() {
			select {
			case p := <-results:
				patch := views.Patch(p.Name, views.PanelBody(p))
				if err := patch.Render(ctx, w); err != nil {
					return err
				}
				if err := Flush().Render(ctx, w); err != nil {
					return err
				}
			case <-ctx.Done():
				return ctx.Err()
			}
		}
		return nil
	})
}

func (s *server) handle(w http.ResponseWriter, r *http.Request) {
	w.Header().Set("Content-Type", "text/html; charset=utf-8")
	// Proxies that buffer would defeat the whole demo; ask them not to.
	w.Header().Set("X-Accel-Buffering", "no")
	page := views.Page(Flush(), s.stream(r.Context()))
	if err := page.Render(r.Context(), w); err != nil {
		// The header is already sent by now, so an error cannot become a 500 —
		// log it and let the truncated response speak for itself.
		log.Printf("render: %v", err)
	}
}
```

```go
package main

import (
	"net/http/httptest"
	"strings"
	"testing"
	"time"
)

// The demo's claim: a later byte fills an earlier hole. Document order is
// revenue, orders, products; completion order is orders, products, revenue.
func TestPatchesArriveInCompletionOrderNotDocumentOrder(t *testing.T) {
	s := &server{latency: map[string]time.Duration{
		"revenue":  60 * time.Millisecond,
		"orders":   10 * time.Millisecond,
		"products": 30 * time.Millisecond,
	}}
	rec := httptest.NewRecorder()
	s.handle(rec, httptest.NewRequest("GET", "/", nil))
	body := rec.Body.String()

	// Regions appear in DOCUMENT order (revenue, orders, products) — the premise
	// the out-of-order fill is measured against. Moved here from the views tests
	// because a full-page render needs a real *vite.Vite in context.
	prev := -1
	for _, name := range []string{"revenue", "orders", "products"} {
		at := strings.Index(body, `<?start name="`+name+`">`)
		if at <= prev {
			t.Fatalf("region %q at %d, not after the previous at %d", name, at, prev)
		}
		prev = at
	}

	// Every region opens before any patch closes it.
	firstPatch := strings.Index(body, "<template for=")
	if firstPatch < 0 {
		t.Fatalf("no patches emitted:\n%s", body)
	}
	for _, name := range []string{"revenue", "orders", "products"} {
		at := strings.Index(body, `<?start name="`+name+`">`)
		if at < 0 || at > firstPatch {
			t.Errorf("region %q at %d must precede the first patch at %d", name, at, firstPatch)
		}
	}

	// Patches arrive fastest-first, which is NOT document order.
	want := []string{"orders", "products", "revenue"}
	var got []string
	for _, seg := range strings.Split(body, `<template for="`)[1:] {
		got = append(got, seg[:strings.IndexByte(seg, '"')])
	}
	if strings.Join(got, ",") != strings.Join(want, ",") {
		t.Errorf("patch order = %v, want %v (completion order, not document order)", got, want)
	}
}

// Every marker the shell opens must be patched exactly once — a region left
// unpatched is a skeleton the user stares at forever.
func TestEveryRegionIsPatchedOnce(t *testing.T) {
```
